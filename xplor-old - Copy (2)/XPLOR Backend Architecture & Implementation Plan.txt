XPLOR Backend Architecture & Implementation Plan
                                        

1. DATABASE ARCHITECTURE
Recommended Approach: PostgreSQL + Redis + S3 (No vector DB initially)
Why this combination:
PostgreSQL (Primary Database):

• ✅ ACID compliance critical for billing/subscriptions

• ✅ JSONB columns handle complex 3D scene data efficiently

• ✅ Perfect for relational data (users → projects → jobs → billing)

• ✅ Complex queries for analytics and plan enforcement

• ✅ Mature ecosystem with Node.js (pg, Prisma, TypeORM)
Redis (Cache + Queue):

• ✅ Session management

• ✅ Job queue (Bull/BullMQ)

• ✅ Rate limiting

• ✅ Real-time editor state caching
S3/Cloud Storage:

• ✅ Floor plan images

• ✅ Generated 3D models

• ✅ VR outputs

• ✅ Asset library files
Vector DB (Future - Phase 3+):

• Add later for smart asset recommendations

• Not critical for MVP

Module 2: Job Tracking System
// services/jobTracker.ts
export class JobTracker {
  async createJob(userId: string, projectId: string, jobType: JobType) {
    // 1. Check plan limits
    const canCreate = await this.checkJobLimit(userId);
    if (!canCreate) throw new Error('Job limit exceeded');
    
    // 2. Create job record
    const job = await prisma.jobs.create({
      data: { user_id: userId, project_id: projectId, job_type: jobType }
    });
    
    // 3. Increment usage counter
    await this.incrementJobUsage(userId);
    
    // 4. Add to queue
    await jobQueue.add(jobType, { jobId: job.id, userId, projectId });
    
    return job;
  }
  
  async trackEditorSession(userId: string, projectId: string) {
    // Track 1-hour sessions
    const session = await prisma.editor_sessions.findFirst({
      where: { user_id: userId, project_id: projectId, is_active: true }
    });



    
    if (!session) {
      // Create new session + job
      const job = await this.createJob(userId, projectId, 'editor_session');
      await prisma.editor_sessions.create({
        data: { user_id: userId, project_id: projectId, job_id: job.id }
      });
    } else {
      // Check if session > 1 hour
      const elapsed = Date.now() - session.started_at.getTime();
      if (elapsed > 3600000) { // 1 hour
        // End current session, create new job
        await this.endSession(session.id);
        await this.trackEditorSession(userId, projectId);
      }
    }
  }
}

Module 3: File Upload Handler
// services/fileUpload.ts
import { S3Client, PutObjectCommand } from '@aws-sdk/client-s3';
import sharp from 'sharp';

export class FileUploadService {
  async uploadFloorPlan(userId: string, file: Express.Multer.File) {
    // 1. Validate file (size, type)
    if (!file.mimetype.startsWith('image/')) {
      throw new Error('Invalid file type');
    }
    
    // 2. Process image (resize, optimize)
    const optimized = await sharp(file.buffer)
      .resize(2048, 2048, { fit: 'inside' })
      .jpeg({ quality: 85 })
      .toBuffer();
    
    // 3. Generate unique filename
    const key = `floor-plans/${userId}/${Date.now()}-${file.originalname}`;
    
    // 4. Upload to S3
    await s3.send(new PutObjectCommand({
      Bucket: process.env.S3_BUCKET,
      Key: key,
      Body: optimized,
      ContentType: 'image/jpeg'
    }));
    
    // 5. Return URL
    return `https://${process.env.S3_BUCKET}.s3.amazonaws.com/${key}`;
  }
}

Module 4: AI Processing Queue



// queues/aiQueue.ts
import { Queue, Worker } from 'bullmq';

const aiQueue = new Queue('ai-processing', {
  connection: { host: 'localhost', port: 6379 }
});

const aiWorker = new Worker('ai-processing', async (job) => {
  const { jobId, floorPlanUrl } = job.data;
  
  // 1. Call FastAPI service
  const response = await fetch('http://ai-service:8000/process-floor-plan', {
    method: 'POST',
    body: JSON.stringify({ image_url: floorPlanUrl }),
    headers: { 'Content-Type': 'application/json' }
  });
  
  const result = await response.json();
  
  // 2. Update job status
  await prisma.jobs.update({
    where: { id: jobId },
    data: { 
      status: 'completed',
      completed_at: new Date(),
      metadata: result
    }
  });
  
  // 3. Store scene data
  await prisma.scenes.create({
    data: {
      project_id: job.data.projectId,
      scene_data: result.scene_json
    }
  });
  
  return result;
});
